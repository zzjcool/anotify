{{t "docs.notify.rules.title"}} 
deliver ⟺ device.enabled AND statusMatch AND tagMatch

statusMatch: device filter all=catch-all · error=errors-only · success=success-only
tagMatch:
  · no deviceTags on notification → broadcast to all enabled devices
  · no tags on device            → receives everything (catch-all)
  · both have tags               → delivers if ≥1 tag in common (ANY)
