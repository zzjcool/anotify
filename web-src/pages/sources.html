apiVersion: anotify/v1
kind: Source
metadata:
  name: gh-advisories
spec:
  kind: poll
  enabled: true
  config:
    url: https://api.github.com/advisories?per_page=20
    schedule:
      expr: "*/5 * * * *"
      tz: UTC
    timeoutMs: 10000
    maxItems: 100
    envelope:
      items: "."
      type:
        pointer: type
      deliveryId:
        pointer: id
      payload:
        whole: true
  targets: []
